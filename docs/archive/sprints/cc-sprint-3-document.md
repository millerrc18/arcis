## Task 5: Import dependency graph

Generate `docs/dependency-graph.md`:

```bash
echo "# Import Dependency Graph" > docs/dependency-graph.md
echo "" >> docs/dependency-graph.md
for f in $(find src -name "*.py" ! -path "*__pycache__*" ! -name "*backup*" | sort); do
    imports=$(grep "^from src\." "$f" | sed 's/from \(src\.[^ ]*\).*/\1/' | sort -u)
    if [ -n "$imports" ]; then
        module=$(echo "$f" | sed 's/\.py$//' | tr '/' '.')
        echo "## $module" >> docs/dependency-graph.md
        echo "$imports" | while read imp; do
            echo "- → $imp" >> docs/dependency-graph.md
        done
        echo "" >> docs/dependency-graph.md
    fi
done
```
